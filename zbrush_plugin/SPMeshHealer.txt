// SPMeshHealer ZBrush Plugin (ZBrush 2021 - 2026+)
// Single-button CGAL mesh & scan repair without hardcoded paths.

[VarDef, dllPath, ""]
[VarDef, dllVer, 0]
[VarDef, exePath, ""]
[VarDef, configPath, ""]
[VarDef, expPath, ""]
[VarDef, impPath, ""]
[VarDef, nl, ""]

[RoutineDef, CheckSystem,
	[VarSet, isMac, [ZBrushInfo, 6]]
	[If, isMac == 1,
		[Note, "SPMeshHealer currently supports Windows only."]
		[Exit]
	]
	[VarSet, dllPath, [FileNameResolvePath, "SPMeshHealerData/ZFileUtils64.dll"]]
	[If, [FileExists, dllPath],
		[VarSet, dllVer, [FileExecute, [Var, dllPath], "Version"]]
		[If, dllVer < 8.0,
			[Note, "ZFileUtils64.dll version is too old."]
			[Exit]
		]
	,
		[Note, "ZFileUtils64.dll not found in SPMeshHealerData."]
		[Exit]
	]
]

[RoutineDef, ResolvePaths,
	[VarSet, nl, [StrMerge, [StrFromAsc, 13], [StrFromAsc, 10]]]
	[VarSet, exePath, [FileNameResolvePath, "SPMeshHealerData/SPMeshHealer.exe"]]
	[VarSet, configPath, [FileNameResolvePath, "SPMeshHealerData/config.txt"]]

	// Determine extension from UI switches (default: GoZ)
	[VarSet, ext, ".goz"]
	[If, [IGet, "ZPlugin:SPMeshHealer:Format OBJ"] == 1, [VarSet, ext, ".obj"]]
	[If, [IGet, "ZPlugin:SPMeshHealer:Format STL"] == 1, [VarSet, ext, ".stl"]]

	[VarSet, expPath, [FileNameResolvePath, [StrMerge, "SPMeshHealerData/SPMeshHealer_in", ext]]]
	[VarSet, impPath, [FileNameResolvePath, [StrMerge, "SPMeshHealerData/SPMeshHealer_out", ext]]]
]

[RoutineDef, WriteConfig,
	[VarSet, hullVal, [IGet, "ZPlugin:SPMeshHealer:Ignore Largest Hole"]]
	[VarSet, pgVal, [IGet, "ZPlugin:SPMeshHealer:Preserve PolyGroups"]]
	[VarSet, quadsVal, [IGet, "ZPlugin:SPMeshHealer:Keep Quads"]]
	[VarSet, fragVal, [IGet, "ZPlugin:SPMeshHealer:Remove Small Fragments"]]
	[VarSet, conVal, [IGet, "ZPlugin:SPMeshHealer:Show Console"]]

	[VarSet, l1, [StrMerge, "hull = ", hullVal, nl]]
	[VarSet, l2, [StrMerge, "preservePolygroups = ", pgVal, nl]]
	[VarSet, l3, [StrMerge, "keepQuads = ", quadsVal, nl]]
	[VarSet, l4, [StrMerge, "removeFragments = ", fragVal, nl]]
	[VarSet, l5, [StrMerge, "showConsole = ", conVal, nl, "minFaces = 50", nl, "minFraction = 0.01", nl, "fixOrientation = 1", nl]]
	[VarSet, l6, [StrMerge, "inFile = ", expPath, nl]]
	[VarSet, l7, [StrMerge, "outFile = ", impPath, nl]]

	[VarSet, cfgLen, [StrLength, l1]]
	[VarAdd, cfgLen, [StrLength, l2]]
	[VarAdd, cfgLen, [StrLength, l3]]
	[VarAdd, cfgLen, [StrLength, l4]]
	[VarAdd, cfgLen, [StrLength, l5]]
	[VarAdd, cfgLen, [StrLength, l6]]
	[VarAdd, cfgLen, [StrLength, l7]]

	[If, [MemGetSize, SPMH_CfgMem], [MemDelete, SPMH_CfgMem]]
	[MemCreate, SPMH_CfgMem, cfgLen, 0]
	[VarSet, off, 0]
	[MemWriteString, SPMH_CfgMem, l1, off, 0] [VarAdd, off, [StrLength, l1]]
	[MemWriteString, SPMH_CfgMem, l2, off, 0] [VarAdd, off, [StrLength, l2]]
	[MemWriteString, SPMH_CfgMem, l3, off, 0] [VarAdd, off, [StrLength, l3]]
	[MemWriteString, SPMH_CfgMem, l4, off, 0] [VarAdd, off, [StrLength, l4]]
	[MemWriteString, SPMH_CfgMem, l5, off, 0] [VarAdd, off, [StrLength, l5]]
	[MemWriteString, SPMH_CfgMem, l6, off, 0] [VarAdd, off, [StrLength, l6]]
	[MemWriteString, SPMH_CfgMem, l7, off, 0]
	[MemSaveToFile, SPMH_CfgMem, configPath, 1]
	[MemDelete, SPMH_CfgMem]
]

[RoutineDef, HealMesh,
	[RoutineCall, CheckSystem]
	[RoutineCall, ResolvePaths]

	[If, [FileExists, exePath], ,
		[Note, [StrMerge, "SPMeshHealer.exe not found at:", nl, exePath]]
		[Exit]
	]

	// Clean stale temp files
	[If, [FileExists, expPath], [FileExecute, [Var, dllPath], "FileDelete", #expPath]]
	[If, [FileExists, impPath], [FileExecute, [Var, dllPath], "FileDelete", #impPath]]

	// 1. Export active SubTool
	[FileNameSetNext, expPath]
	[IPress, Tool:Export]
	[If, [FileExists, expPath], ,
		[Note, [StrMerge, "Failed to export mesh to:", nl, expPath]]
		[Exit]
	]

	// 2. Write config with temp paths & options, then launch SPMeshHealer.exe
	[RoutineCall, WriteConfig]
	[If, [FileExists, configPath], ,
		[Note, [StrMerge, "Failed to write config file:", nl, configPath]]
		[Exit]
	]
	[VarSet, err, [FileExecute, [Var, dllPath], "LaunchAppWithFile", #exePath]]

	// 3. Wait for atomic output file (up to 300 seconds)
	[VarSet, found, 0]
	[Loop, 3000,
		[If, [FileExists, impPath],
			[VarSet, found, 1]
			[LoopExit]
		]
		[Delay, 0.1]
	]

	// 4. Import healed mesh & clean up temp files
	[If, found == 1,
		[FileNameSetNext, impPath]
		[IPress, Tool:Import]
		[NoteBar, "SPMeshHealer: Mesh healed successfully."]
	,
		[Note, "SPMeshHealer timed out or failed to produce output."]
	]

	[If, [FileExists, expPath], [FileExecute, [Var, dllPath], "FileDelete", #expPath]]
	[If, [FileExists, impPath], [FileExecute, [Var, dllPath], "FileDelete", #impPath]]
]

[ISubPalette, "ZPlugin:SPMeshHealer"]

	[RoutineCall, HealMesh]

	[IEnable, "ZPlugin:SPMeshHealer:Ignore Largest Hole"]
,
	[IEnable, "ZPlugin:SPMeshHealer:Ignore Largest Hole"]

	[IEnable, "ZPlugin:SPMeshHealer:Remove Small Fragments"]

	[IEnable, "ZPlugin:SPMeshHealer:Show Console"]

	[IUnPress, "ZPlugin:SPMeshHealer:Format OBJ"]
	[IUnPress, "ZPlugin:SPMeshHealer:Format STL"]

	[IUnPress, "ZPlugin:SPMeshHealer:Format STL"]

	[IUnPress, "ZPlugin:SPMeshHealer:Format GoZ"]
